# A face theme for glass-evo.
#
# A theme is a folder with this file in it. Every line is optional: what a
# theme leaves out keeps the built-in look, and what the user sets in the
# Glass Manager (Screen tab, "How the screen looks") goes over the theme, key
# for key. A key a face does not know is passed over, so a theme written
# for a later glass-evo still draws on an earlier one.
#
# Install: copy the folder to /data/INTERNAL/glass/faces/ on the player and
# choose it on the Manager's Screen tab. A theme there stands before a look
# of the same name that glass-evo ships.

[theme]
# The theme's name, as its folder's, and a line about it: both for the
# list the look is chosen from.
name = Example
description = The built-in look, written out

[colours]
# The colour of glass and the colour of what is lit: "artwork" takes each
# from the cover of what plays, or fix one as #rrggbb.
tint = artwork
accent = artwork
# Text and icons.
ink = #f2f2f5

[glass]
# How much of the glass shows over the picture, 0.1 to 1.
bar = 0.75
sheet = 0.78
# The line along a glass's inner edge, 0 to 1.
hairline = 0.12
# Frost what lies under the glass while the theme moves: auto (as suits the
# board, which the plugin knows), on, or off.
frost = auto

[buttons]
# The icons on their own: a colour, or "ink" for the theme's ink.
ink = ink
opacity = 1.0

[clock]
# The clock shows when nothing plays: on or off.
show = on
# The time as a pattern (strftime's): %H:%M is 13:05, %H:%M:%S adds the
# seconds, %-I:%M %p is 1:05 PM.
format = %H:%M
# The clock on its own: a colour, or "ink" for the theme's ink.
ink = ink
opacity = 0.86
# How much of the glass behind the clock and the date shows; 0 for no glass.
glass = 0.55
# That glass's colour: "tint" for the theme's tint, or a colour of its own.
tint = tint

[date]
# The date when nothing plays: on or off. Its place: "top" of the screen,
# on a glass of its own, or "above" or "below" the clock, on the clock's.
show = off
place = top
# The date as a pattern: %A %-d %B is Thursday 1 October, %a %-d %b %Y is
# Thu 1 Oct 2026, %d/%m/%Y is 01/10/2026, %Y-%m-%d is 2026-10-01.
format = %A %-d %B
ink = ink
opacity = 0.86
# The glass behind a date at the top of the screen; 0 for no glass.
glass = 0.55
tint = tint

[measure]
# Heights in units of a 720th of the picture's height; the face size
# (normal, large, car) scales them together.
bar = 72
clock = 144
date = 40
