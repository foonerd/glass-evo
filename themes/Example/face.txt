# A face theme for glass-evo.
#
# A theme is a folder with this file in it. Every line is optional: what a
# theme leaves out keeps the built-in look, and what the user sets in the
# Glass Manager (Screen tab, "How the screen looks") goes over the theme, key
# for key. A key a face does not know is passed over, so a theme written
# for a later glass-evo still draws on an earlier one.
#
# Install: copy the folder to /data/INTERNAL/glass/faces/ on the player and
# choose it on the Manager's Screen tab. A theme there stands before a look
# of the same name that glass-evo ships.
#
# A meter theme may bring a look of its own: a face.txt of this form beside
# its meters.txt. While that theme is on show its keys go over the look
# chosen in the Manager, key for key, and what the user sets there still
# goes over both. It needs no installing and no choosing: it comes and goes
# with the theme.

[theme]
# The theme's name, as its folder's, and a line about it: both for the
# list the look is chosen from.
name = Example
description = The built-in look, written out

[colours]
# The colour of glass and the colour of what is lit: "artwork" takes each
# from the cover of what plays, or fix one as #rrggbb.
tint = artwork
accent = artwork
# Text and icons.
ink = #f2f2f5

[glass]
# How much of the glass shows over the picture, 0.1 to 1.
bar = 0.75
sheet = 0.78
# The line along a glass's inner edge, 0 to 1.
hairline = 0.12
# Frost what lies under the glass while the theme moves: auto (as suits the
# board, which the plugin knows), on, or off.
frost = auto

[buttons]
# The icons on their own: a colour, or "ink" for the theme's ink.
ink = ink
opacity = 1.0

[clock]
# The clock shows when nothing plays: on or off.
show = on
# Where it stands, on the idle screen's grid: three rows (top, middle,
# bottom) by three columns (left, centre, right) in equal thirds above the
# bar. place names the cells it occupies, rows then columns, each one name
# or a range: "middle left-right" is the middle row, "middle-bottom
# centre-right" four cells, "top-bottom left-right" all nine. Empty: in the
# middle of what the date and the bar leave, as before the grid. align is
# where it stands inside its cells: left, centre or right, and top, middle
# or bottom. A size the look comes with is fitted to the cells; one the
# user set is kept, and runs over them where it is larger. margin is the
# room, in units of a 720th of the picture's height, between its glass and
# the sides of its cells it is aligned to; 0 puts it in the corner.
place =
align = centre middle
margin = 20
# The time as a pattern (strftime's): %H:%M is 13:05, %H:%M:%S adds the
# seconds, %-I:%M %p is 1:05 PM.
format = %H:%M
# The clock on its own: a colour, or "ink" for the theme's ink.
ink = ink
opacity = 0.86
# How much of the glass behind the clock and the date shows; 0 for no glass.
glass = 0.55
# That glass's colour: "tint" for the theme's tint, or a colour of its own.
tint = tint
# The clock's face: "type" sets the time in the theme's type; "seven" and
# "sixteen" draw it in segments, "flip" on cards that fall, "dial" with
# hands. A drawn face is as high as measure.clock, a dial twice that.
face = type
# A dial's style: "station" (bars and strong hands), "numbers", "roman", or
# "plain" (marks alone). It has a second hand when the format shows seconds.
dial = station
# How much of an unlit segment shows, 0 to 0.5.
unlit = 0.08
# A dial's hands, and its marks and numerals: a colour, or "ink" for the
# clock's ink (a station dial has dark ones of its own).
hands = ink
marks = ink
# The second hand: a colour, or "accent" for the look's (red on a station dial).
second = accent
# The disc behind a dial's hands: "style" for the style's own (a station
# dial has a light one, the others none), "none", or a colour.
disc = style
# A flip clock's cards.
card = #17171a

[date]
# The date when nothing plays: on or off. Its place: "top" of the screen,
# on a glass of its own, or "above" or "below" the clock, on the clock's.
show = off
place = top
# The date as a pattern: %A %-d %B is Thursday 1 October, %a %-d %b %Y is
# Thu 1 Oct 2026, %d/%m/%Y is 01/10/2026, %Y-%m-%d is 2026-10-01.
format = %A %-d %B
ink = ink
opacity = 0.86
# The glass behind a date at the top of the screen; 0 for no glass.
glass = 0.55
tint = tint

[measure]
# Heights in units of a 720th of the picture's height; the face size
# (normal, large, car) scales them together.
bar = 72
clock = 144
date = 40

[idle]
# When nothing plays: a picture of the user's own behind the clock and the
# date, in the theme's place. The name of a file in the player's
# glass/backgrounds folder (JPEG, PNG or WebP), chosen on the Manager's
# Screen tab; empty for none, and the theme shows. It is scaled to cover the
# screen and cut from its middle. dim darkens it, 0 to 0.9.
picture =
dim = 0.25
# Minutes with nothing playing and no touch after which the screen goes
# black, for a screen that must not show the same picture for hours; a tap
# wakes it, and so does music. 0: never.
off = 0
# Milliseconds the screen takes to go black and to come back, up to 120000
# (two minutes); 0 for at once.
fade = 500
